```ts
import { CheckBox } from "@neneys/ui";

CheckBox('#id', { 
    trailing: ...,

})
```

## Parameters

| Param | Value |
|:---------|:------:|
| trailing | Widget |
| value | boolean |

## Example

```ts
import { CheckBox, Text } from "@neneys/ui";

CheckBox('#id', { 
    trailing: Text('#txt', {text: "Checkbox"}),
    value: false
})
```
